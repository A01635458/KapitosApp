CREATE TABLE public.conversations (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  client_id uuid NOT NULL,
  producer_id uuid NOT NULL,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  last_message_at timestamp without time zone,
  is_active boolean DEFAULT true,
  CONSTRAINT conversations_pkey PRIMARY KEY (id),
  CONSTRAINT conversations_client_id_fkey FOREIGN KEY (client_id) REFERENCES public.profiles(id),
  CONSTRAINT conversations_producer_id_fkey FOREIGN KEY (producer_id) REFERENCES public.profiles(id)
);
CREATE TABLE public.messages (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  conversation_id uuid NOT NULL,
  sender_id uuid NOT NULL,
  content text NOT NULL,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  is_read boolean DEFAULT false,
  read_at timestamp without time zone,
  is_deleted boolean DEFAULT false,
  deleted_at timestamp without time zone,
  message_type text DEFAULT 'text'::text CHECK (message_type = ANY (ARRAY['text'::text, 'image'::text, 'system'::text])),
  image_url text,
  metadata jsonb,
  CONSTRAINT messages_pkey PRIMARY KEY (id),
  CONSTRAINT messages_sender_id_fkey FOREIGN KEY (sender_id) REFERENCES public.profiles(id),
  CONSTRAINT messages_conversation_id_fkey FOREIGN KEY (conversation_id) REFERENCES public.conversations(id)
);
CREATE TABLE public.producer_products (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  producer_id uuid NOT NULL,
  name text NOT NULL,
  price numeric NOT NULL CHECK (price >= 0::numeric),
  weight text NOT NULL,
  image_url text,
  description text,
  tasting_notes ARRAY,
  brewing_methods ARRAY,
  category text,
  roast_level text,
  packaging_type text,
  category_fields jsonb,
  is_active boolean DEFAULT true,
  created_at timestamp without time zone DEFAULT now(),
  updated_at timestamp without time zone DEFAULT now(),
  CONSTRAINT producer_products_pkey PRIMARY KEY (id),
  CONSTRAINT producer_products_producer_id_fkey FOREIGN KEY (producer_id) REFERENCES public.profiles(id)
);
CREATE TABLE public.producers (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  farm_name text,
  experience_years integer,
  created_at timestamp without time zone DEFAULT now(),
  phone text,
  photo_url text,
  farm_size_ha numeric CHECK (farm_size_ha IS NULL OR farm_size_ha >= 0::numeric),
  country text,
  state text,
  municipality text,
  annual_production_kg integer,
  last_harvest_date date,
  yield_per_ha numeric,
  price_per_kg numeric CHECK (price_per_kg IS NULL OR price_per_kg >= 0::numeric),
  min_contract_volume integer,
  open_to_export boolean DEFAULT false,
  sells_online boolean DEFAULT false,
  online_store_url text,
  has_tourist_area boolean DEFAULT false,
  tourist_accessible boolean DEFAULT false,
  tourism_details text,
  consent_gps boolean DEFAULT false,
  consent_ai boolean DEFAULT false,
  consent_notifications boolean DEFAULT false,
  varieties ARRAY,
  processes ARRAY,
  certifications ARRAY,
  altitude integer,
  status text DEFAULT 'pending'::text CHECK (status = ANY (ARRAY['pending'::text, 'approved'::text, 'rejected'::text])),
  latitude numeric CHECK (latitude IS NULL OR latitude >= '-90'::integer::numeric AND latitude <= 90::numeric),
  longitude numeric CHECK (longitude IS NULL OR longitude >= '-180'::integer::numeric AND longitude <= 180::numeric),
  shade_coverage_percent integer CHECK (shade_coverage_percent IS NULL OR shade_coverage_percent >= 0 AND shade_coverage_percent <= 100),
  sales_types ARRAY,
  CONSTRAINT producers_pkey PRIMARY KEY (id)
);
CREATE TABLE public.profiles (
  id uuid NOT NULL,
  full_name text NOT NULL,
  email text NOT NULL UNIQUE,
  role text DEFAULT 'user'::text CHECK (role = ANY (ARRAY['user'::text, 'producer'::text, 'admin'::text])),
  created_at timestamp without time zone DEFAULT now(),
  photo_url text,
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
CREATE TABLE public.unread_messages_count (
  user_id uuid NOT NULL,
  conversation_id uuid NOT NULL,
  unread_count integer DEFAULT 0,
  last_updated timestamp without time zone DEFAULT now(),
  CONSTRAINT unread_messages_count_pkey PRIMARY KEY (user_id, conversation_id),
  CONSTRAINT unread_messages_count_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id),
  CONSTRAINT unread_messages_count_conversation_id_fkey FOREIGN KEY (conversation_id) REFERENCES public.conversations(id)
);
CREATE TABLE public.user_preferences (
  user_id uuid NOT NULL,
  processes ARRAY,

  times ARRAY,

  created_at timestamp without time zone DEFAULT now(),

  CONSTRAINT user_preferences_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id)
);
