  image_id bigint,
  bean_type text,
  defects jsonb,
  quality_score numeric,
  ai_explanation text,
  created_at timestamp without time zone DEFAULT now(),

  roast_level text CHECK (roast_level = ANY (ARRAY['light'::text, 'medium'::text, 'dark'::text, 'unroasted'::text])),
  uniformity_score numeric CHECK (uniformity_score >= 0::numeric AND uniformity_score <= 100::numeric),
  color_analysis jsonb,
  defect_count integer DEFAULT 0,

  CONSTRAINT ai_inspections_image_id_fkey FOREIGN KEY (image_id) REFERENCES public.coffee_images(id)
